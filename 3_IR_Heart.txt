
# 3) Heart Disease Diagnosis using Bayesian Network

import pandas as pd
from pgmpy.models import DiscreteBayesianNetwork
from pgmpy.estimators import BayesianEstimator
from pgmpy.inference import VariableElimination

# Load Heart Disease Dataset
data = pd.read_csv("Heart_disease_cleveland.csv")

print("Dataset Loaded Successfully!")
print(data.head())

# Select attributes
data = data[["age", "cp", "chol", "thalach",
             "exang", "target"]].copy()

# Convert values to numeric
data = data.apply(pd.to_numeric, errors="coerce")
data = data.dropna()

# Convert target into 0 and 1
data["target"] = (data["target"] > 0).astype(int)

# Convert continuous attributes into categories
for col in ["age", "chol", "thalach"]:
    data[col] = pd.cut(
        data[col],
        bins=3,
        labels=False
    )

data = data.dropna()
data = data.astype(int)

# Create Bayesian Network
model = DiscreteBayesianNetwork([
    ("age", "target"),
    ("cp", "target"),
    ("chol", "target"),
    ("thalach", "target"),
    ("exang", "target")
])

# Train the model using Bayesian Estimator
estimator = BayesianEstimator(model, data)

cpds = estimator.get_parameters(prior_type="BDeu")
model.add_cpds(*cpds)

print("\nModel trained successfully!")
print("Model validation:", model.check_model())

# Perform inference
inference = VariableElimination(model)

result = inference.query(
    variables=["target"],
    evidence={
        "age": 2,
        "cp": 3,
        "chol": 2,
        "thalach": 0,
        "exang": 1
    }
)

print("\nHeart Disease Diagnosis:")
print(result)
