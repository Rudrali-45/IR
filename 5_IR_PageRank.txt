# 5)Problem Statement: Implement Page Rank Algorithm

# Page Rank Algorithm

pages = ['A', 'B', 'C', 'D']

# Initial PageRank
rank = {
    'A': 0.25,
    'B': 0.25,
    'C': 0.25,
    'D': 0.25
}

# Links between pages
links = {
    'A': [],
    'B': ['C', 'A'],
    'C': ['A'],
    'D': ['A', 'B', 'C']
}

# Calculate PageRank for first iteration
new_rank = {page: 0 for page in pages}

for page in pages:
    if len(links[page]) > 0:
        share = rank[page] / len(links[page])

        for linked_page in links[page]:
            new_rank[linked_page] += share

print("PageRank after First Iteration:")

for page in pages:
    print(page, ":", round(new_rank[page], 3))
